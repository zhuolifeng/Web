======== 8. 反范式收益：我的订单列表 ========
-- 范式化：JOIN users + JOIN order_items + GROUP BY
mysql> EXPLAIN SELECT o.order_no, u.nickname, SUM(i.quantity) AS item_count, GROUP_CONCAT(i.book_title) AS titles, o.total_amount, o.created_at FROM orders o JOIN users u ON u.id = o.user_id JOIN order_items i ON i.order_id = o.id WHERE o.user_id = 42 GROUP BY o.id ORDER BY o.created_at DESC LIMIT 10
